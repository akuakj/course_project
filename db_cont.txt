
from PySide6.QtWidgets import (QTableWidgetItem, QMessageBox, QHeaderView,
                               QPushButton, QHBoxLayout, QWidget)
import soundfile as sf
import os
from .db_manager import TinyDBVoiceManager


class DatabaseController:
    def __init__(self, main_controller):
        self.main = main_controller
        self.db_manager = TinyDBVoiceManager()
        self.current_people = []

    def setup_connections(self):
        """Подключение кнопок базы данных"""
        self.main.btn_refresh_db.clicked.connect(self.refresh_database)
        self.main.btn_delete_record.clicked.connect(self.delete_selected_record)
        self.main.btn_search_text.clicked.connect(self.search_in_database)

        # Автоматически обновляем при переходе на вкладку
        self.main.btn_database.clicked.connect(self.refresh_database)

    def refresh_database(self):
        """Обновление таблицы базы данных с кнопками проигрывания"""
        try:
            # Получаем все записи из БД
            self.current_people = self.db_manager.get_all_people()

            # Настраиваем таблицу
            table = self.main.table_voices
            table.setRowCount(len(self.current_people))

            # Очищаем старые кнопки
            for row in range(table.rowCount()):
                cell_widget = table.cellWidget(row, 3)  # Столбец с кнопками
                if cell_widget:
                    cell_widget.deleteLater()

            # Заполняем таблицу данными
            for row, person in enumerate(self.current_people):
                # Порядковый номер (1, 2, 3...)
                table.setItem(row, 0, QTableWidgetItem(str(row + 1)))

                # ФИО
                table.setItem(row, 1, QTableWidgetItem(person['full_name']))

                # Дата добавления (форматируем)
                created_date = person['created_at']
                if 'T' in created_date:
                    date_part = created_date.split('T')[0]
                    time_part = created_date.split('T')[1][:8]
                    formatted_date = f"{date_part} {time_part}"
                else:
                    formatted_date = created_date
                table.setItem(row, 2, QTableWidgetItem(formatted_date))

                # Столбец с кнопками проигрывания
                self._add_playback_buttons(row, person)

            # Настраиваем заголовки таблицы
            header = table.horizontalHeader()
            header.setSectionResizeMode(0, QHeaderView.ResizeMode.ResizeToContents)  # №
            header.setSectionResizeMode(1, QHeaderView.ResizeMode.Stretch)  # ФИО
            header.setSectionResizeMode(2, QHeaderView.ResizeMode.ResizeToContents)  # Дата
            header.setSectionResizeMode(3, QHeaderView.ResizeMode.ResizeToContents)  # Управление

            # Обновляем статистику
            self.update_statistics()

            print(f"База данных обновлена: {len(self.current_people)} записей")

        except Exception as e:
            print(f"Ошибка обновления базы данных: {e}")
            QMessageBox.warning(self.main, "Ошибка", f"Не удалось загрузить базу данных: {e}")

    def play_audio_file(self, audio_file_path):
        """Проигрывание аудио файла"""
        try:
            if not os.path.exists(audio_file_path):
                QMessageBox.warning(self.main, "Ошибка", f"Файл не найден:\n{audio_file_path}")
                return

            # Используем analysis_controller для проигрывания
            if hasattr(self.main, 'analysis_controller'):
                analysis_controller = self.main.analysis_controller

                # Останавливаем текущее воспроизведение
                analysis_controller.stop_audio()

                # Загружаем файл напрямую через soundfile
                try:
                    import soundfile as sf
                    audio_data, sample_rate = sf.read(audio_file_path)

                    # Если аудио многоканальное - преобразуем в моно
                    if audio_data.ndim > 1:
                        audio_data = audio_data.mean(axis=1)

                    # Устанавливаем данные в контроллер
                    analysis_controller.current_audio_file = audio_file_path
                    analysis_controller.audio_data = audio_data
                    analysis_controller.sample_rate = sample_rate
                    analysis_controller.audio_length = len(audio_data) / sample_rate
                    analysis_controller.current_position = 0
                    analysis_controller.current_sample = 0

                    # Обновляем интерфейс
                    file_name = os.path.basename(audio_file_path)
                    self.main.label_file_name.setText(f"Файл: {file_name}")
                    self.main.progress_audio.setValue(0)

                    # Активируем кнопки управления
                    self.main.btn_start_audio.setEnabled(True)
                    self.main.btn_pause_audio.setEnabled(True)
                    self.main.btn_close_audio.setEnabled(True)

                    # Запускаем проигрывание
                    analysis_controller.start_audio()

                    # Переключаемся на вкладку анализа чтобы видеть прогресс
                    self.main.stackedWidget.setCurrentIndex(1)

                    # Показываем уведомление
                    QMessageBox.information(self.main, "Проигрывание", f"Проигрывается: {file_name}")

                except Exception as load_error:
                    print(f"Ошибка загрузки аудио файла: {load_error}")
                    QMessageBox.warning(self.main, "Ошибка", f"Не удалось загрузить аудио файл: {load_error}")

            else:
                QMessageBox.warning(self.main, "Ошибка", "Модуль воспроизведения недоступен")

        except Exception as e:
            print(f"Ошибка проигрывания файла: {e}")
            QMessageBox.warning(self.main, "Ошибка", f"Ошибка при проигрывании: {e}")

    def show_file_info(self, audio_file_path):
        """Показ информации о файле"""
        try:
            if os.path.exists(audio_file_path):
                file_name = os.path.basename(audio_file_path)
                file_size = os.path.getsize(audio_file_path) / 1024  # Размер в KB
                file_info = f"Файл: {file_name}\nРазмер: {file_size:.1f} KB\nПуть: {audio_file_path}"
                QMessageBox.information(self.main, "Информация о файле", file_info)
            else:
                QMessageBox.warning(self.main, "Ошибка", "Файл не найден")
        except Exception as e:
            print(f"Ошибка показа информации о файле: {e}")

    def show_all_files(self, audio_files):
        """Показ всех файлов для записи"""
        try:
            if len(audio_files) <= 1:
                return

            files_info = "Доступные файлы:\n\n"
            for i, file_path in enumerate(audio_files, 1):
                file_name = os.path.basename(file_path)
                files_info += f"{i}. {file_name}\n"

            QMessageBox.information(self.main, "Все файлы записи", files_info)
        except Exception as e:
            print(f"Ошибка показа всех файлов: {e}")

    def update_statistics(self):
        """Обновление статистики базы данных"""
        try:
            stats = self.db_manager.get_statistics()

            self.main.label_total_records.setText(f"Всего записей: {stats['total_records']}")
            self.main.label_last_update.setText(f"Последнее обновление: {stats['last_update']}")

            # Обновляем статус БД
            if stats['total_records'] > 0:
                self.main.label_db_status.setText("Статус БД: ✅ OK")
                self.main.label_db_status.setStyleSheet("""
                    QLabel {
                        color: #27AE60;
                        font-weight: bold;
                        padding: 5px 10px;
                        background-color: #EAFAEE;
                        border: 1px solid #27AE60;
                        border-radius: 3px;
                    }
                """)
            else:
                self.main.label_db_status.setText("Статус БД: ⚠️ Пусто")
                self.main.label_db_status.setStyleSheet("""
                    QLabel {
                        color: #F39C12;
                        font-weight: bold;
                        padding: 5px 10px;
                        background-color: #FEF9E7;
                        border: 1px solid #F39C12;
                        border-radius: 3px;
                    }
                """)

        except Exception as e:
            print(f"Ошибка обновления статистики: {e}")

    def delete_selected_record(self):
        """Удаление выбранной записи"""
        try:
            table = self.main.table_voices
            selected_rows = table.selectionModel().selectedRows()

            if not selected_rows:
                QMessageBox.information(self.main, "Информация", "Выберите запись для удаления")
                return

            selected_row = selected_rows[0].row()

            if selected_row >= len(self.current_people):
                QMessageBox.warning(self.main, "Ошибка", "Неверный выбор записи")
                return

            # Получаем реальный UUID из данных
            person_data = self.current_people[selected_row]
            person_id = person_data['id']
            person_name = person_data['full_name']

            # Подтверждение удаления
            reply = QMessageBox.question(
                self.main,
                "Подтверждение удаления",
                f"Вы уверены, что хотите удалить запись:\n{person_name}?",
                QMessageBox.StandardButton.Yes | QMessageBox.StandardButton.No
            )

            if reply == QMessageBox.StandardButton.Yes:
                success = self.db_manager.delete_person(person_id)
                if success:
                    QMessageBox.information(self.main, "Успех", "Запись успешно удалена")
                    self.refresh_database()
                else:
                    QMessageBox.warning(self.main, "Ошибка", "Не удалось удалить запись")

        except Exception as e:
            print(f"Ошибка удаления записи: {e}")
            QMessageBox.warning(self.main, "Ошибка", f"Ошибка при удалении: {e}")

    def search_in_database(self):
        """Поиск по базе данных"""
        try:
            search_text = self.main.lineEdit_search.text().strip().lower()

            if not search_text:
                self.refresh_database()  # Показываем все если поиск пустой
                return

            all_people = self.db_manager.get_all_people()
            filtered_people = []

            for person in all_people:
                # Ищем в ФИО и заметках
                if (search_text in person['full_name'].lower() or
                        search_text in person.get('notes', '').lower()):
                    filtered_people.append(person)

            # Сохраняем текущий список для отображения
            self.current_people = filtered_people

            # Обновляем таблицу с результатами поиска
            table = self.main.table_voices
            table.setRowCount(len(filtered_people))

            for row, person in enumerate(filtered_people):
                # Порядковый номер
                table.setItem(row, 0, QTableWidgetItem(str(row + 1)))

                # ФИО
                table.setItem(row, 1, QTableWidgetItem(person['full_name']))

                # Дата
                created_date = person['created_at']
                if 'T' in created_date:
                    date_part = created_date.split('T')[0]
                    formatted_date = date_part
                else:
                    formatted_date = created_date
                table.setItem(row, 2, QTableWidgetItem(formatted_date))

                # Вектор (полная информация)
                vector_data = person['vector_data']
                vector_info = f"Размер: {len(vector_data)} | "
                vector_info += f"Min: {min(vector_data):.4f} | "
                vector_info += f"Max: {max(vector_data):.4f} | "
                vector_info += f"Mean: {sum(vector_data) / len(vector_data):.4f}"

                table.setItem(row, 3, QTableWidgetItem(vector_info))

            # Обновляем статистику для поиска
            self.main.label_total_records.setText(f"Найдено записей: {len(filtered_people)}")

            if not filtered_people:
                QMessageBox.information(self.main, "Поиск", "Записи не найдены")

        except Exception as e:
            print(f"Ошибка поиска: {e}")
            QMessageBox.warning(self.main, "Ошибка", f"Ошибка при поиске: {e}")

    def get_selected_person_id(self):
        """Получение ID выбранного человека"""
        table = self.main.table_voices
        selected_rows = table.selectionModel().selectedRows()

        if selected_rows and self.current_people:
            selected_row = selected_rows[0].row()
            if selected_row < len(self.current_people):
                return self.current_people[selected_row]['id']
        return None
